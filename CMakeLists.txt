cmake_minimum_required(VERSION 3.16)
project(gtetrinet-ng C)

set(GTN_VERSION "0.0.0-dev" CACHE STRING "Version shown in the game and sent to servers")
# Off for source and development builds: only published release binaries post anonymous faults.
option(GTN_FAULT_REPORTS "Let the game post anonymous fault reports" OFF)
option(GTN_BUILD_GAME "Build the SDL2 game (off: core, server and tests only)" ON)
option(BUILD_TESTING "Build the tests" ON)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  add_compile_options(-Wall -Wextra -Wno-unused-parameter)
endif()

if(GTN_FAULT_REPORTS)
  set(GTN_FAULT_REPORTS_VALUE 1)
else()
  set(GTN_FAULT_REPORTS_VALUE 0)
endif()
configure_file(cmake/version.h.in ${CMAKE_CURRENT_BINARY_DIR}/generated/version.h @ONLY)

add_library(tncore STATIC
  src/core/tn_protocol.c src/core/tn_game.c src/core/tn_net.c
  src/core/tn_client.c src/core/tn_server.c src/core/tn_bot.c)
target_include_directories(tncore PUBLIC src/core ${CMAKE_CURRENT_BINARY_DIR}/generated)
if(WIN32)
  target_link_libraries(tncore PUBLIC ws2_32)
endif()

add_executable(gtetrinet-ng-server src/server/main.c)
target_link_libraries(gtetrinet-ng-server tncore)

if(GTN_BUILD_GAME)
  find_package(SDL2 2.0.14 REQUIRED)
  add_executable(gtetrinet-ng WIN32
    src/app/main.c src/app/render.c src/app/input.c src/app/ui.c src/app/fault.c src/app/settings.c
    src/app/i18n.c src/app/sound.c)
  target_include_directories(gtetrinet-ng PRIVATE src/app third_party)
  if(TARGET SDL2::SDL2main)
    target_link_libraries(gtetrinet-ng SDL2::SDL2main)
  endif()
  if(TARGET SDL2::SDL2)
    target_link_libraries(gtetrinet-ng SDL2::SDL2)
  else()
    target_include_directories(gtetrinet-ng PRIVATE ${SDL2_INCLUDE_DIRS})
    target_link_libraries(gtetrinet-ng ${SDL2_LIBRARIES})
  endif()
  target_link_libraries(gtetrinet-ng tncore)
  if(NOT WIN32)
    target_link_libraries(gtetrinet-ng m)
  endif()
  if(BUILD_TESTING)
    foreach(input_test test_input test_input_virtual test_effects test_repeat test_i18n test_controls test_names test_sound)
      add_executable(${input_test} tests/${input_test}.c src/app/render.c src/app/input.c src/app/ui.c src/app/fault.c
        src/app/i18n.c src/app/sound.c)
      if(NOT input_test MATCHES "^test_(repeat|i18n|controls|names|sound)$")
        target_sources(${input_test} PRIVATE src/app/settings.c)
      endif()
      target_include_directories(${input_test} PRIVATE src/app third_party ${SDL2_INCLUDE_DIRS})
      target_link_libraries(${input_test} tncore)
      if(TARGET SDL2::SDL2)
        target_link_libraries(${input_test} SDL2::SDL2)
      else()
        target_link_libraries(${input_test} ${SDL2_LIBRARIES})
      endif()
      if(NOT WIN32)
        target_link_libraries(${input_test} m)
      endif()
    endforeach()
  endif()
  install(TARGETS gtetrinet-ng RUNTIME DESTINATION bin)
endif()
install(TARGETS gtetrinet-ng-server RUNTIME DESTINATION bin)

if(BUILD_TESTING)
  enable_testing()
  add_executable(test_core tests/test_core.c)
  target_link_libraries(test_core tncore)
  add_test(NAME core COMMAND test_core)
  set_tests_properties(core PROPERTIES TIMEOUT 60)
  if(GTN_BUILD_GAME)
    add_test(NAME input COMMAND test_input)
    add_test(NAME input_virtual COMMAND test_input_virtual)
    add_test(NAME effects COMMAND test_effects)
    file(MAKE_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/repeat-prefs)
    add_test(NAME repeat COMMAND test_repeat "${CMAKE_CURRENT_BINARY_DIR}/repeat-prefs/")
    file(MAKE_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/i18n-prefs)
    add_test(NAME i18n COMMAND test_i18n "${CMAKE_CURRENT_BINARY_DIR}/i18n-prefs/")
    file(MAKE_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/controls-prefs)
    add_test(NAME controls COMMAND test_controls "${CMAKE_CURRENT_BINARY_DIR}/controls-prefs/")
    file(MAKE_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/names-prefs)
    add_test(NAME names COMMAND test_names "${CMAKE_CURRENT_BINARY_DIR}/names-prefs/")
    file(MAKE_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/sound-prefs)
    add_test(NAME sound COMMAND test_sound "${CMAKE_CURRENT_BINARY_DIR}/sound-prefs/")
    set_tests_properties(input input_virtual effects repeat i18n controls names sound PROPERTIES TIMEOUT 30 ENVIRONMENT "GTN_NO_REPORT=1")
  endif()
endif()

install(FILES CHANGELOG.md COPYING DESTINATION share/gtetrinet-ng)
